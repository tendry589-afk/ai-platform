🗄️ SCHÉMA PostgreSQL (Données Relationnelles)

1. Tables Utilisateurs

-- Table Users (Utilisateurs)
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    username VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100),
    last_name VARCHAR(100),
    avatar_url TEXT,
    bio TEXT,
    phone_number VARCHAR(20),
    is_active BOOLEAN DEFAULT true,
    is_verified BOOLEAN DEFAULT false,
    is_superuser BOOLEAN DEFAULT false,
    last_login TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP,
    CONSTRAINT email_format CHECK (email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Z|a-z]{2,}$')
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_created_at ON users(created_at);

-- Table User Profiles (Profils Utilisateurs)
CREATE TABLE user_profiles (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID UNIQUE NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    subscription_tier VARCHAR(50) DEFAULT 'free', -- free, pro, enterprise
    subscription_expires_at TIMESTAMP,
    preferences JSONB DEFAULT '{}',
    settings JSONB DEFAULT '{}',
    language VARCHAR(10) DEFAULT 'en',
    timezone VARCHAR(50) DEFAULT 'UTC',
    theme VARCHAR(20) DEFAULT 'light',
    notifications_enabled BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_user_profiles_user_id ON user_profiles(user_id);
CREATE INDEX idx_user_profiles_subscription_tier ON user_profiles(subscription_tier);

-- Table API Keys (Clés API)
CREATE TABLE api_keys (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    key_hash VARCHAR(255) UNIQUE NOT NULL,
    name VARCHAR(100) NOT NULL,
    description TEXT,
    last_used_at TIMESTAMP,
    expires_at TIMESTAMP,
    is_active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_api_keys_user_id ON api_keys(user_id);
CREATE INDEX idx_api_keys_key_hash ON api_keys(key_hash);


2. Tables Chatbot

-- Table Conversations (Conversations)
CREATE TABLE conversations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    title VARCHAR(255),
    description TEXT,
    model_version VARCHAR(50) DEFAULT 'gpt2-base',
    language VARCHAR(10) DEFAULT 'en',
    temperature FLOAT DEFAULT 0.7,
    max_tokens INT DEFAULT 200,
    system_prompt TEXT,
    is_archived BOOLEAN DEFAULT false,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP
);

CREATE INDEX idx_conversations_user_id ON conversations(user_id);
CREATE INDEX idx_conversations_created_at ON conversations(created_at);

-- Table Messages (Messages)
CREATE TABLE messages (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    conversation_id UUID NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    content TEXT NOT NULL,
    role VARCHAR(20) NOT NULL, -- 'user' or 'assistant'
    embedding VECTOR(384), -- Embeddings pour recherche sémantique
    tokens_used INT,
    processing_time_ms INT,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_messages_conversation_id ON messages(conversation_id);
CREATE INDEX idx_messages_user_id ON messages(user_id);
CREATE INDEX idx_messages_role ON messages(role);
CREATE INDEX idx_messages_embedding ON messages USING ivfflat (embedding vector_cosine_ops);


3. Tables Recommandations

-- Table Recommendations (Recommandations)
CREATE TABLE recommendations (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    item_id VARCHAR(255) NOT NULL,
    item_type VARCHAR(50) NOT NULL, -- 'product', 'article', 'user', 'content'
    item_title VARCHAR(255),
    item_description TEXT,
    item_metadata JSONB DEFAULT '{}',
    score FLOAT NOT NULL,
    confidence FLOAT,
    reason TEXT,
    algorithm VARCHAR(50), -- 'collaborative', 'content-based', 'hybrid'
    is_clicked BOOLEAN DEFAULT false,
    is_purchased BOOLEAN DEFAULT false,
    feedback_score INT, -- 1-5 rating
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expires_at TIMESTAMP,
    deleted_at TIMESTAMP
);

CREATE INDEX idx_recommendations_user_id ON recommendations(user_id);
CREATE INDEX idx_recommendations_item_id ON recommendations(item_id);
CREATE INDEX idx_recommendations_score ON recommendations(score DESC);
CREATE INDEX idx_recommendations_created_at ON recommendations(created_at);

-- Table User Interactions (Interactions Utilisateurs)
CREATE TABLE user_interactions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    item_id VARCHAR(255) NOT NULL,
    item_type VARCHAR(50) NOT NULL,
    interaction_type VARCHAR(50) NOT NULL, -- 'view', 'click', 'like', 'purchase', 'share'
    interaction_value FLOAT DEFAULT 1.0,
    duration_seconds INT,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_user_interactions_user_id ON user_interactions(user_id);
CREATE INDEX idx_user_interactions_item_id ON user_interactions(item_id);
CREATE INDEX idx_user_interactions_type ON user_interactions(interaction_type);
CREATE INDEX idx_user_interactions_created_at ON user_interactions(created_at);


4. Tables Prédictions

-- Table Predictions (Prédictions)
CREATE TABLE predictions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    model_name VARCHAR(100) NOT NULL,
    model_version VARCHAR(50),
    input_data JSONB NOT NULL,
    prediction JSONB NOT NULL,

    prediction_type VARCHAR(50), -- 'regression', 'classification', 'clustering'

    feedback TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_predictions_user_id ON predictions(user_id);
CREATE INDEX idx_predictions_model_name ON predictions(model_name);
CREATE INDEX idx_predictions_created_at ON predictions(created_at);

-- Table Training Data (Données d'Entraînement)
CREATE TABLE training_data (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    model_name VARCHAR(100) NOT NULL,
    features JSONB NOT NULL,
    target JSONB NOT NULL,
    split VARCHAR(20) DEFAULT 'train', -- 'train', 'validation', 'test'

CREATE INDEX idx_training_data_model_name ON training_data(model_name);
CREATE INDEX idx_training_data_split ON training_data(split);


5. Tables Classification d'Images

-- Table Image Classifications (Classifications d'Images)
CREATE TABLE image_classifications (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    image_url TEXT NOT NULL,
    image_hash VARCHAR(255),
    image_size_bytes INT,
    image_dimensions JSONB, -- {width, height}
    model_name VARCHAR(100) NOT NULL,
    model_version VARCHAR(50),
    classifications JSONB NOT NULL, -- [{label, confidence, bounding_box}]
    processing_time_ms INT,
    confidence_threshold FLOAT DEFAULT 0.5,
    metadata JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_image_classifications_user_id ON image_classifications(user_id);
CREATE INDEX idx_image_classifications_model_name ON image_classifications(model_name);
CREATE INDEX idx_image_classifications_created_at ON image_classifications(created_at);

-- Table Image Detections (Détections d'Objets)
CREATE TABLE image_detections (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    classification_id UUID NOT NULL REFERENCES image_classifications(id) ON DELETE CASCADE,
    object_label VARCHAR(100) NOT NULL,
    confidence FLOAT NOT NULL,
    bounding_box JSONB, -- {x, y, width, height}
    area_percentage FLOAT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_image_detections_classification_id ON image_detections(classification_id);
CREATE INDEX idx_image_detections_object_label ON image_detections(object_label);

🔐 CONTRAINTES & VALIDATIONS

-- Contraintes de domaine
ALTER TABLE users ADD CONSTRAINT check_email_format 
    CHECK (email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\\.[A-Z|a-z]{2,}$');

ALTER TABLE user_profiles ADD CONSTRAINT check_subscription_tier 
    CHECK (subscription_tier IN ('free', 'pro', 'enterprise'));

ALTER TABLE messages ADD CONSTRAINT check_role 
    CHECK (role IN ('user', 'assistant'));

ALTER TABLE recommendations ADD CONSTRAINT check_score_range 
    CHECK (score >= 0 AND score <= 1);

ALTER TABLE predictions ADD CONSTRAINT check_confidence_range 
    CHECK (confidence >= 0 AND confidence <= 1);

-- Contraintes d'unicité
ALTER TABLE api_keys ADD CONSTRAINT unique_key_hash UNIQUE (key_hash);
ALTER TABLE ai_models ADD CONSTRAINT unique_model_version UNIQUE (name, version);

-- Contraintes de clés étrangères avec cascade
ALTER TABLE user_profiles ADD CONSTRAINT fk_user_profiles_user_id 
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE;

ALTER TABLE conversations ADD CONSTRAINT fk_conversations_user_id 
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE;

ALTER TABLE messages ADD CONSTRAINT fk_messages_conversation_id 
    FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE;




📈 STRATÉGIE DE PARTITIONNEMENT

-- Partitionnement par date pour les tables volumineuses
CREATE TABLE messages_partitioned (
    id UUID,
    conversation_id UUID,
    user_id UUID,
    content TEXT,
    role VARCHAR(20),
    created_at TIMESTAMP
) PARTITION BY RANGE (created_at);

-- Créer des partitions mensuelles
CREATE TABLE messages_2024_01 PARTITION OF messages_partitioned
    FOR VALUES FROM ('2024-01-01') TO ('2024-02-01');

CREATE TABLE messages_2024_02 PARTITION OF messages_partitioned
    FOR VALUES FROM ('2024-02-01') TO ('2024-03-01');

-- Partitionnement par hash pour les tables de recommandations
CREATE TABLE recommendations_partitioned (
    id UUID,
    user_id UUID,
    item_id VARCHAR(255),
    score FLOAT
) PARTITION BY HASH (user_id);

CREATE TABLE recommendations_part_0 PARTITION OF recommendations_partitioned
    FOR VALUES WITH (MODULUS 4, REMAINDER 0);

CREATE TABLE recommendations_part_1 PARTITION OF recommendations_partitioned
    FOR VALUES WITH (MODULUS 4, REMAINDER 1);
