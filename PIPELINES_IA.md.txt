# backend/app/ai/nlp/chatbot.py

from transformers import AutoTokenizer, AutoModel, pipeline
from sentence_transformers import SentenceTransformer
import torch
import numpy as np
from typing import List, Dict, Tuple
import logging

logger = logging.getLogger(__name__)

class ChatbotEngine:
    def __init__(self, model_name: str = "gpt2", embedding_model: str = "all-MiniLM-L6-v2"):
        """Initialiser le moteur chatbot"""
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        
        # Modèles
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModel.from_pretrained(model_name).to(self.device)
        self.embedding_model = SentenceTransformer(embedding_model)
        
        # Pipelines
        self.text_generation = pipeline(
            "text-generation",
            model=model_name,
            device=0 if torch.cuda.is_available() else -1
        )
        self.ner = pipeline("ner", model="dslim/bert-base-NER")
        self.sentiment = pipeline("sentiment-analysis")
        
        logger.info(f"Chatbot initialized on {self.device}")
    
    def preprocess_text(self, text: str) -> Dict:
        """Prétraiter le texte utilisateur"""
        # Normalisation
        text = text.strip().lower()
        
        # Tokenization
        tokens = self.tokenizer.encode(text, return_tensors="pt").to(self.device)
        
        return {
            "original": text,
            "tokens": tokens,
            "token_count": tokens.shape[1]
        }
    
    def generate_embedding(self, text: str) -> np.ndarray:
        """Générer l'embedding du texte"""
        embedding = self.embedding_model.encode(text, convert_to_numpy=True)
        return embedding / np.linalg.norm(embedding)  # Normalisation
    
    def retrieve_context(self, embedding: np.ndarray, db_session, k: int = 5) -> List[Dict]:
        """Récupérer le contexte similaire"""
        from sqlalchemy import func
        from app.models import Message
        
        # Recherche sémantique dans pgvector
        similar_messages = db_session.query(Message).order_by(
            Message.embedding.cosine_distance(embedding)
        ).limit(k).all()
        
        context = []
        for msg in similar_messages:
            context.append({
                "role": msg.role,
                "content": msg.content,
                "similarity": 1 - msg.embedding.cosine_distance(embedding)
            })
        
        return context
    
    def classify_intent(self, text: str) -> Dict:
        """Classifier l'intention de l'utilisateur"""
        # NER
        entities = self.ner(text)
        
        # Sentiment
        sentiment = self.sentiment(text)[0]
        
        # Topics (simple keyword matching)
        topics = self._extract_topics(text)
        
        return {
            "entities": entities,
            "sentiment": sentiment,
            "topics": topics
        }
    
    def generate_response(self, text: str, context: List[Dict], max_length: int = 200) -> str:
        """Générer la réponse"""
        # Construire le prompt avec contexte
        prompt = self._build_prompt(text, context)
        
        # Générer
        response = self.text_generation(
            prompt,
            max_length=max_length,
            num_return_sequences=1,
            temperature=0.7,
            top_p=0.95,
            do_sample=True
        )
        
        return response[0]["generated_text"]
    
    def post_process_response(self, response: str) -> str:
        """Post-traiter la réponse"""
        # Correction grammaticale (simple)
        response = response.strip()
        
        # Filtrage de toxicité
        if self._is_toxic(response):
            response = "I appreciate your question, but I can't provide that response."
        
        return response
    
    async def chat(self, user_id: str, conversation_id: str, message: str, db_session) -> Dict:
        """Pipeline complet de chat"""
        try:
            # 1. Prétraitement
            preprocessed = self.preprocess_text(message)
            
            # 2. Embedding
            embedding = self.generate_embedding(message)
            
            # 3. Récupération du contexte
            context = self.retrieve_context(embedding, db_session)
            
            # 4. Classification d'intention
            intent = self.classify_intent(message)
            
            # 5. Génération de réponse
            response = self.generate_response(message, context)
            
            # 6. Post-traitement
            response = self.post_process_response(response)
            
            # 7. Stockage
            from app.models import Message
            user_msg = Message(
                conversation_id=conversation_id,
                user_id=user_id,
                content=message,
                role="user",
                embedding=embedding
            )
            assistant_msg = Message(
                conversation_id=conversation_id,
                content=response,
                role="assistant",
                embedding=self.generate_embedding(response)
            )
            db_session.add(user_msg)
            db_session.add(assistant_msg)
            db_session.commit()
            
            return {
                "response": response,
                "intent": intent,
                "context_used": len(context),
                "processing_time_ms": 150
            }
        
        except Exception as e:
            logger.error(f"Chat error: {str(e)}")
            raise
    
    def _build_prompt(self, text: str, context: List[Dict]) -> str:
        """Construire le prompt avec contexte"""
        prompt = "Previous conversation:\n"
        for ctx in context:
            prompt += f"{ctx['role']}: {ctx['content']}\n"
        prompt += f"User: {text}\nAssistant:"
        return prompt
    
    def _extract_topics(self, text: str) -> List[str]:
        """Extraire les topics du texte"""
        # Implémentation simple
        keywords = ["weather", "sports", "technology", "health", "finance"]
        topics = [kw for kw in keywords if kw in text.lower()]
        return topics
    
    def _is_toxic(self, text: str) -> bool:
        """Vérifier si le texte est toxique"""
        # Implémentation simple
        toxic_words = ["hate", "kill", "bad"]
        return any(word in text.lower() for word in toxic_words)

# backend/app/ai/ml/recommender.py

import numpy as np
from sklearn.decomposition import NMF
from sklearn.metrics.pairwise import cosine_similarity
from typing import List, Dict, Tuple
import logging

logger = logging.getLogger(__name__)

class RecommendationEngine:
    def __init__(self, n_factors: int = 50):
        """Initialiser le moteur de recommandations"""
        self.n_factors = n_factors
        self.nmf_model = NMF(n_components=n_factors, random_state=42)
        self.user_factors = None
        self.item_factors = None
    
    def build_interaction_matrix(self, interactions: List[Dict]) -> Tuple[np.ndarray, Dict, Dict]:
        """Construire la matrice d'interactions"""
        user_ids = sorted(set(i["user_id"] for i in interactions))
        item_ids = sorted(set(i["item_id"] for i in interactions))
        
        user_idx = {uid: i for i, uid in enumerate(user_ids)}
        item_idx = {iid: i for i, iid in enumerate(item_ids)}
        
        # Créer la matrice
        matrix = np.zeros((len(user_ids), len(item_ids)))
        
        for interaction in interactions:
            u_idx = user_idx[interaction["user_id"]]
            i_idx = item_idx[interaction["item_id"]]
            matrix[u_idx, i_idx] = interaction.get("weight", 1.0)
        
        return matrix, user_idx, item_idx
    
    def train(self, interaction_matrix: np.ndarray):
        """Entraîner le modèle"""
        logger.info("Training NMF model...")
        self.user_factors = self.nmf_model.fit_transform(interaction_matrix)
        self.item_factors = self.nmf_model.components_.T
        logger.info("Training completed")
    
    def get_recommendations(self, user_id: str, user_idx: Dict, item_idx: Dict, 
                           k: int = 10, exclude_items: List[str] = None) -> List[Dict]:
        """Obtenir les recommandations pour un utilisateur"""
        if user_id not in user_idx:
            return []
        
        u_idx = user_idx[user_id]
        user_factor = self.user_factors[u_idx]
        
        # Calculer les scores
        scores = np.dot(user_factor, self.item_factors.T)
        
        # Exclure les items déjà vus
        if exclude_items:
            for item_id in exclude_items:
                if item_id in item_idx:
                    scores[item_idx[item_id]] = -np.inf
        
        # Top-K
        top_indices = np.argsort(scores)[-k:][::-1]
        
        # Construire les résultats
        idx_to_item = {v: k for k, v in item_idx.items()}
        recommendations = []
        
        for idx in top_indices:
            if scores[idx] > -np.inf:
                recommendations.append({
                    "item_id": idx_to_item[idx],
                    "score": float(scores[idx]),
                    "algorithm": "collaborative_filtering"
                })
        
        return recommendations
    
    def content_based_recommendations(self, item_id: str, item_features: np.ndarray, 
                                     k: int = 10) -> List[Dict]:
        """Recommandations basées sur le contenu"""
        # Calculer la similarité cosinus
        similarities = cosine_similarity([item_features])[0]
        
        # Top-K
        top_indices = np.argsort(similarities)[-k-1:-1][::-1]
        
        recommendations = []
        for idx in top_indices:
            recommendations.append({
                "item_id": idx,
                "score": float(similarities[idx]),
                "algorithm": "content_based"
            })
        
        return recommendations
    
    def hybrid_recommendations(self, user_id: str, item_id: str, 
                              user_idx: Dict, item_idx: Dict,
                              item_features: np.ndarray,
                              k: int = 10, alpha: float = 0.5) -> List[Dict]:
        """Recommandations hybrides (collaborative + content)"""
        # Collaborative
        collab_recs = self.get_recommendations(user_id, user_idx, item_idx, k)
        
        # Content-based
        content_recs = self.content_based_recommendations(item_id, item_features, k)
        
        # Fusion
        combined = {}
        for rec in collab_recs:
            combined[rec["item_id"]] = alpha * rec["score"]
        
        for rec in content_recs:
            if rec["item_id"] in combined:
                combined[rec["item_id"]] += (1 - alpha) * rec["score"]
            else:
                combined[rec["item_id"]] = (1 - alpha) * rec["score"]
        
        # Trier
        sorted_recs = sorted(combined.items(), key=lambda x: x[1], reverse=True)
        
        return [
            {"item_id": item_id, "score": score, "algorithm": "hybrid"}
            for item_id, score in sorted_recs[:k]
        ]
    
    async def get_user_recommendations(self, user_id: str, db_session, k: int = 10) -> List[Dict]:
        """Pipeline complet de recommandations"""
        try:
            # 1. Récupérer les interactions
            from app.models import UserInteraction
            interactions = db_session.query(UserInteraction).all()
            
            # 2. Construire la matrice
            matrix, user_idx, item_idx = self.build_interaction_matrix(
                [{"user_id": i.user_id, "item_id": i.item_id, "weight": i.interaction_value} 
                 for i in interactions]
            )
            
            # 3. Entraîner (si nécessaire)
            if self.user_factors is None:
                self.train(matrix)
            
            # 4. Générer les recommandations
            recommendations = self.get_recommendations(user_id, user_idx, item_idx, k)
            
            # 5. Appliquer les règles métier
            recommendations = self._apply_business_rules(recommendations)
            
            # 6. Sauvegarder
            from app.models import Recommendation
            for rec in recommendations:
                db_rec = Recommendation(
                    user_id=user_id,
                    item_id=rec["item_id"],
                    score=rec["score"],
                    algorithm=rec["algorithm"]
                )
                db_session.add(db_rec)
            db_session.commit()
            
            return recommendations
        
        except Exception as e:
            logger.error(f"Recommendation error: {str(e)}")
            raise
    
    def _apply_business_rules(self, recommendations: List[Dict]) -> List[Dict]:
        """Appliquer les règles métier"""
        # Diversité
        recommendations = self._ensure_diversity(recommendations)
        
        # Freshness boost
        recommendations = self._boost_freshness(recommendations)
        
        return recommendations
    
    def _ensure_diversity(self, recommendations: List[Dict]) -> List[Dict]:
        """Assurer la diversité des résultats"""
        # Implémentation simple
        return recommendations[:10]
    
    def _boost_freshness(self, recommendations: List[Dict]) -> List[Dict]:
        """Booster les items récents"""
        # Implémentation simple
        return recommendations

# backend/app/ai/ml/predictor.py

import numpy as np
import pandas as pd
from sklearn.preprocessing import StandardScaler, LabelEncoder
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.ensemble import RandomForestRegressor
from xgboost import XGBRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error, r2_score
import joblib
import logging
from typing import Dict, List, Tuple, Any

logger = logging.getLogger(__name__)

class PredictionEngine:
    def __init__(self, model_type: str = "xgboost"):
        """Initialiser le moteur de prédiction"""
        self.model_type = model_type
        self.model = None
        self.scaler = StandardScaler()
        self.label_encoders = {}
        self.feature_names = None
        self.metrics = {}
    
    def prepare_data(self, df: pd.DataFrame, target_column: str) -> Tuple[np.ndarray, np.ndarray]:
        """Préparer les données"""
        logger.info("Preparing data...")
        
        # Copie
        df = df.copy()
        
        # Gestion des valeurs manquantes
        df = df.fillna(df.mean(numeric_only=True))
        
        # Séparation features/target
        X = df.drop(columns=[target_column])
        y = df[target_column]
        
        # Encoding des variables catégoriques
        for col in X.select_dtypes(include=['object']).columns:
            le = LabelEncoder()
            X[col] = le.fit_transform(X[col])
            self.label_encoders[col] = le
        
        # Normalisation
        X_scaled = self.scaler.fit_transform(X)
        
        self.feature_names = X.columns.tolist()
        
        logger.info(f"Data prepared: {X_scaled.shape}")
        return X_scaled, y.values
    
    def train(self, X: np.ndarray, y: np.ndarray, test_size: float = 0.2):
        """Entraîner le modèle"""
        logger.info(f"Training {self.model_type} model...")
        
        # Split
        X_train, X_test, y_train, y_test = train_test_split(
            X, y, test_size=test_size, random_state=42
        )
        
        # Créer le modèle
        if self.model_type == "xgboost":
            self.model = XGBRegressor(
                n_estimators=100,
                max_depth=6,
                learning_rate=0.1,
                random_state=42,
                n_jobs=-1
            )
        else:
            self.model = RandomForestRegressor(
                n_estimators=100,
                max_depth=10,
                random_state=42,
                n_jobs=-1
            )
        
        # Entraîner
        self.model.fit(X_train, y_train)
        
        # Évaluer
        y_pred = self.model.predict(X_test)
        
        self.metrics = {
            "mae": mean_absolute_error(y_test, y_pred),
            "rmse": np.sqrt(mean_squared_error(y_test, y_pred)),
            "r2": r2_score(y_test, y_pred)
        }
        
        logger.info(f"Model trained. Metrics: {self.metrics}")
    
    def predict(self, X: np.ndarray) -> Dict[str, Any]:
        """Faire une prédiction"""
        if self.model is None:
            raise ValueError("Model not trained yet")
        
        # Prédiction
        prediction = self.model.predict(X)[0]
        
        # Confiance (basée sur la variance des arbres)
        if hasattr(self.model, 'estimators_'):
            predictions = np.array([tree.predict(X)[0] for tree in self.model.estimators_])
            confidence = 1 - (np.std(predictions) / np.mean(predictions))
        else:
            confidence = 0.8
        
        # Feature importance
        feature_importance = dict(zip(
            self.feature_names,
            self.model.feature_importances_
        ))
        
        return {
            "prediction": float(prediction),
            "confidence": float(np.clip(confidence, 0, 1)),
            "feature_importance": feature_importance,
            "metrics": self.metrics
        }
    
    def save_model(self, path: str):
        """Sauvegarder le modèle"""
        joblib.dump(self.model, f"{path}/model.pkl")
        joblib.dump(self.scaler, f"{path}/scaler.pkl")
        joblib.dump(self.label_encoders, f"{path}/encoders.pkl")
        logger.info(f"Model saved to {path}")
    
    def load_model(self, path: str):
        """Charger le modèle"""
        self.model = joblib.load(f"{path}/model.pkl")
        self.scaler = joblib.load(f"{path}/scaler.pkl")
        self.label_encoders = joblib.load(f"{path}/encoders.pkl")
        logger.info(f"Model loaded from {path}")
    
    async def predict_for_user(self, user_id: str, input_data: Dict, db_session) -> Dict:
        """Pipeline complet de prédiction"""
        try:
            # 1. Préparer les données
            df = pd.DataFrame([input_data])
            X = self.scaler.transform(df)
            
            # 2. Prédire
            result = self.predict(X)
            
            # 3. Sauvegarder
            from app.models import Prediction
            prediction = Prediction(
                user_id=user_id,
                model_name=self.model_type,
                input_data=input_data,
                prediction=result,
                confidence=result["confidence"]
            )
            db_session.add(prediction)
            db_session.commit()
            
            return result
        
        except Exception as e:
            logger.error(f"Prediction error: {str(e)}")
            raise

# backend/app/ai/cv/classifier.py

import cv2
import numpy as np
from PIL import Image
import torch
import torchvision.transforms as transforms
from torchvision.models import resnet50, detection
import logging
from typing import Dict, List, Tuple
import io

logger = logging.getLogger(__name__)

class ImageClassifier:
    def __init__(self, model_name: str = "resnet50"):
        """Initialiser le classificateur d'images"""
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.model_name = model_name
        
        # Charger le modèle
        if model_name == "resnet50":
            self.model = resnet50(pretrained=True).to(self.device)
        else:
            self.model = resnet50(pretrained=True).to(self.device)
        
        self.model.eval()
        
        # Transformations
        self.transforms = transforms.Compose([
            transforms.Resize((224, 224)),
            transforms.ToTensor(),
            transforms.Normalize(
                mean=[0.485, 0.456, 0.406],
                std=[0.229, 0.224, 0.225]
            )
        ])
        
        # Classes ImageNet
        self.classes = self._load_imagenet_classes()
        
        logger.info(f"Image classifier initialized on {self.device}")
    
    def preprocess_image(self, image_path: str) -> torch.Tensor:
        """Prétraiter l'image"""
        image = Image.open(image_path).convert('RGB')
        image_tensor = self.transforms(image).unsqueeze(0).to(self.device)
        return image_tensor
    
    def classify(self, image_path: str, top_k: int = 5) -> List[Dict]:
        """Classifier l'image"""
        # Prétraitement
        image_tensor = self.preprocess_image(image_path)
        
        # Inférence
        with torch.no_grad():
            outputs = self.model(image_tensor)
        
        # Top-K
        probabilities = torch.nn.functional.softmax(outputs, dim=1)
        top_probs, top_indices = torch.topk(probabilities, top_k)
        
        # Résultats
        results = []
        for prob, idx in zip(top_probs[0], top_indices[0]):
            results.append({
                "label": self.classes[idx.item()],
                "confidence": float(prob.item())
            })
        
        return results
    
    def detect_objects(self, image_path: str) -> List[Dict]:
        """Détecter les objets dans l'image"""
        # Charger l'image
        image = cv2.imread(image_path)
        image_rgb = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
        
        # Prétraitement
        image_tensor = transforms.ToTensor()(Image.fromarray(image_rgb)).unsqueeze(0).to(self.device)
        
        # Inférence (utiliser un modèle de détection)
        # Pour cet exemple, on utilise une approche simple
        detections = self._simple_detection(image_rgb)
        
        return detections
    
    def _simple_detection(self, image: np.ndarray) -> List[Dict]:
        """Détection simple basée sur les contours"""
        gray = cv2.cvtColor(image, cv2.COLOR_RGB2GRAY)
        edges = cv2.Canny(gray, 100, 200)
        contours, _ = cv2.findContours(edges, cv2.RETR_TREE, cv2.CHAIN_APPROX_SIMPLE)
        
        detections = []
        for contour in contours[:10]:  # Top 10
            x, y, w, h = cv2.boundingRect(contour)
            area = w * h
            
            if area > 100:  # Filtrer les petits contours
                detections.append({
                    "label": "object",
                    "confidence": 0.8,
                    "bounding_box": {
                        "x": int(x),
                        "y": int(y),
                        "width": int(w),
                        "height": int(h)
                    },
                    "area_percentage": float((area / (image.shape[0] * image.shape[1])) * 100)
                })
        
        return detections
    
    async def classify_image_for_user(self, user_id: str, image_path: str, db_session) -> Dict:
        """Pipeline complet de classification"""
        try:
            # 1. Classifier
            classifications = self.classify(image_path)
            
            # 2. Détecter les objets
            detections = self.detect_objects(image_path)
            
            # 3. Sauvegarder
            from app.models import ImageClassification
            img_class = ImageClassification(
                user_id=user_id,
                image_url=image_path,
                model_name=self.model_name,
                classifications=classifications,
                model_version="1.0.0"
            )
            db_session.add(img_class)
            db_session.commit()
            
            return {
                "classifications": classifications,
                "detections": detections,
                "processing_time_ms": 200
            }
        
        except Exception as e:
            logger.error(f"Classification error: {str(e)}")
            raise
    
    def _load_imagenet_classes(self) -> List[str]:
        """Charger les classes ImageNet"""
        # Implémentation simplifiée
        return [f"class_{i}" for i in range(1000)]
